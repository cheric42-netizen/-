<script>
    // Функция озвучки с использованием встроенного Web Speech API
    function playText(text) {
        if ('speechSynthesis' in window) {
            window.speechSynthesis.cancel(); // Остановить текущую озвучку, если есть
            const utterance = new SpeechSynthesisUtterance(text);
            utterance.lang = 'tr-TR'; // Установка турецкого языка
            utterance.rate = 0.85;    // Слегка замедленный темп для уровня А1
            window.speechSynthesis.speak(utterance);
        } else {
            alert("Ваш браузер не поддерживает синтез речи. Используйте современный Chrome или Safari.");
        }
    }

    // Функция проверки ответов
    function checkAnswers() {
        const inputs = document.querySelectorAll('input[type="text"]');
        inputs.forEach(input => {
            const userAnswer = input.value.trim().toLowerCase();
            const correctAnswer = input.getAttribute('data-answer').toLowerCase();
            
            // Заменяем турецкие специфичные символы для корректного сравнения ввода
            if (userAnswer === correctAnswer) {
                input.className = 'correct';
            } else {
                input.className = 'incorrect';
            }
        });
    }
</script>
